function Get-Fibonacci {
    [OutputType([bigint])]
    param(
        [Parameter(Mandatory)]
        [ValidateRange(0, [int]::MaxValue)]
        [int]$N
    )

    [bigint]$current = 0
    [bigint]$next = 1

    for ($index = 0; $index -lt $N; $index++) {
        [bigint]$following = $current + $next
        $current = $next
        $next = $following
    }

    return $current
}
